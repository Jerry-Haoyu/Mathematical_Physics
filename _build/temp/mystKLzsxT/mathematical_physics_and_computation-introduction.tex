\subsubsection{Definition}

\begin{definition}A PDE for a function $u(x_1,...,x_n)$ is of the form:
\begin{equation}
F(x_1,...,x_n, u,u_{x_1},...,u_{x_n},u_{x_1x_1},u_{x_1x_2},...)=0
\end{equation}

\begin{enumerate}
\item The \textit{order} is the order of the highest-order derivative
\item An equation is called \textit{linear} if it is linear on $u$ and its derivatives
\item An equation is callled \textit{semi-linear} if all coefficinets are only functions of $x_1,...,x_n$
\item An equation is called \textit{quasilinear} if coefficinet $a_N$ is only functions of $x,u$ and lower-order derivatives.
\end{enumerate}

\end{definition}
