% Declares a numbered, colored box (used for exercises and every `prf:`

% ==========================================================================
% Chapter headings
% --------------------------------------------------------------------------
% The project's top-level table-of-contents groups (e.g. "Calculus of
% Variation") compile to \section (not \chapter — see myst-to-tex's heading
% depth mapping), so by default they show as small numbered run-in headings
% like "0.2 Calculus of Variation" (the "0." is the never-incremented
% \chapter counter). Restyle \section as a centered "Chapter <roman>" page
% instead, and drop the leftover chapter-number prefix from \thesection so
% nested numbering reads "II.1" rather than "0.2.1".
\makeatletter
\renewcommand{\thesection}{\arabic{section}}
\renewcommand{\section}[1]{%
  \clearpage
  \refstepcounter{section}%
  \phantomsection
  \addcontentsline{toc}{section}{\protect\numberline{\Roman{section}}#1}%
  \begingroup
    \centering
    \vspace*{2.5cm}
    {\Large\scshape Chapter \Roman{section}}\\[1em]
    {\huge\bfseries #1}\par
    \vspace{2cm}
  \endgroup
}
\makeatother
